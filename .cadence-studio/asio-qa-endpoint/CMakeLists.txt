cmake_minimum_required(VERSION 3.22)
project(CadenceQaClock VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(CADENCE_QA_PORTABLE_METRICS_ONLY "Build arithmetic tests only; never a Windows qualification" OFF)
enable_testing()
add_executable(cadence_qa_metrics_test metrics_test.cpp)
add_test(NAME portable_output_metrics COMMAND cadence_qa_metrics_test)
if(CADENCE_QA_PORTABLE_METRICS_ONLY)
    return()
endif()
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The endpoint and ABI harness require native Windows x64 MSVC. Use portable mode only for arithmetic tests.")
endif()
add_library(CadenceQaClock SHARED qa_clock.cpp qa_clock.def)
add_executable(CadenceQaClockHarness harness.cpp)
foreach(target CadenceQaClock CadenceQaClockHarness)
    target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS _WIN32_WINNT=0x0A00 WINVER=0x0A00)
    target_compile_options(${target} PRIVATE /W4 /EHsc)
    target_link_libraries(${target} PRIVATE ole32 uuid)
endforeach()
