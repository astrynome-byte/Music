# Reuse the original WiX module over an immutable installed directory.
# There is no application target, compiler invocation or install rule here.
cmake_minimum_required(VERSION 3.22)
project(CadenceRetained2008Packaging LANGUAGES NONE)
if(NOT CMAKE_HOST_WIN32 OR NOT WIN32)
    message(FATAL_ERROR "Actual packaging requires native Windows")
endif()
if(NOT CADENCE_STAGE_ROOT OR NOT IS_DIRECTORY "${CADENCE_STAGE_ROOT}/bin")
    message(FATAL_ERROR "An authenticated existing stage is required")
endif()
get_filename_component(CADENCE_STAGE_ROOT "${CADENCE_STAGE_ROOT}" ABSOLUTE)
if(NOT CADENCE_PRODUCT_VERSION STREQUAL "4.7.6.2008")
    message(FATAL_ERROR "Only the actual retained 2008 product may be packaged")
endif()
file(SHA256 "${CADENCE_STAGE_ROOT}/bin/CadenceStudio4.exe" _cadence_exe_sha)
if(NOT _cadence_exe_sha STREQUAL "13cd4e6d2f408d2e64b187d4c3bff1188ff4192ac7b670d3a12373bbeb37f6cd")
    message(FATAL_ERROR "Actual retained 2008 executable bytes differ")
endif()
if(EXISTS "${CADENCE_STAGE_ROOT}/bin/icuuc.dll" OR IS_SYMLINK "${CADENCE_STAGE_ROOT}/bin/icuuc.dll")
    message(FATAL_ERROR "Local ICU forwarding DLL must remain absent")
endif()
# Native host/process and AMD64 PE machine were checked by the wrapper.
# This is installer architecture metadata, not a simulated application build.
set(PLATFORM_DETECTED 1)
set(OS_IS_WIN 1)
set(CMAKE_SIZEOF_VOID_P 8)
set(CPACK_WIX_ARCHITECTURE x64)
set(CADENCE_BUILD ON)
set(CMAKE_BUILD_NUMBER 2008)
set(MUSE_APP_BUILD_MODE release)
include("${PROJECT_SOURCE_DIR}/version.cmake")
set(MUSE_APP_UNSTABLE OFF)
set(MUSE_APP_IS_PRERELEASE OFF)
list(PREPEND CMAKE_MODULE_PATH "${PROJECT_SOURCE_DIR}/buildscripts/cmake")
# All redistributed DLLs are already authenticated stage files; never replace
# them with another developer machine's CRT or Qt installation.
set(CMAKE_INSTALL_SYSTEM_RUNTIME_LIBS_SKIP TRUE)
set(CMAKE_INSTALL_PREFIX "${CADENCE_STAGE_ROOT}" CACHE PATH "Immutable retained stage" FORCE)
set(MUE_PACK_INSTALL_DIR ON CACHE BOOL "Package only the verified stage" FORCE)
set(CPACK_WIX_TEMPLATE "${PROJECT_SOURCE_DIR}/buildscripts/packaging/Windows/Installer/WIX.template.in")
include("${PROJECT_SOURCE_DIR}/buildscripts/packaging/Windows/SetupWindowsPackaging.cmake")
if(NOT CPACK_PACKAGE_VERSION STREQUAL CADENCE_PRODUCT_VERSION OR CPACK_INSTALL_CMAKE_PROJECTS)
    message(FATAL_ERROR "Packaging must have exact product version and no install project")
endif()
if(NOT CPACK_WIX_PRODUCT_GUID STREQUAL "CA4AC56B-9ED9-568F-9F5A-0D9475D4CC26"
   OR NOT CPACK_WIX_UPGRADE_GUID STREQUAL "319B992F-470A-465B-9A79-6B0CBFC7D6C7")
    message(FATAL_ERROR "Independent Cadence installer identities differ")
endif()
if(NOT CPACK_INSTALLED_DIRECTORIES STREQUAL "${CADENCE_STAGE_ROOT};.")
    message(FATAL_ERROR "Only the exact authenticated stage may enter the MSI")
endif()
