# Package an already built application stage. There is deliberately no target.
cmake_minimum_required(VERSION 3.22)
project(CadenceRetainedStagePackaging LANGUAGES CXX)
if(NOT CADENCE_SOURCE_ROOT OR NOT CADENCE_STAGE_ROOT OR NOT CMAKE_BUILD_NUMBER STREQUAL "2002")
    message(FATAL_ERROR "An audited source root, existing stage and original build number 2002 are required")
endif()
if(NOT WIN32 AND NOT CADENCE_RETAINED_CONFIGURE_ONLY)
    message(FATAL_ERROR "Actual packaging requires native Windows; configuration-only is not an MSI test")
endif()
if(WIN32 AND NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The packaging compiler/Windows SDK environment must target x64")
endif()
string(REPLACE "\\" "/" CADENCE_SOURCE_ROOT "${CADENCE_SOURCE_ROOT}")
string(REPLACE "\\" "/" CADENCE_STAGE_ROOT "${CADENCE_STAGE_ROOT}")
if(NOT EXISTS "${CADENCE_STAGE_ROOT}/bin/CadenceStudio4.exe")
    message(FATAL_ERROR "The verified application stage is missing")
endif()
list(PREPEND CMAKE_MODULE_PATH "${CADENCE_SOURCE_ROOT}/buildscripts/cmake")
set(CADENCE_BUILD ON CACHE BOOL "Use the exact Cadence source packaging metadata" FORCE)
set(MUSE_APP_BUILD_MODE release)
include("${CADENCE_SOURCE_ROOT}/version.cmake")
set(MUSE_APP_UNSTABLE OFF)
set(MUSE_APP_IS_PRERELEASE OFF)
if(CADENCE_RETAINED_CONFIGURE_ONLY)
    set(PLATFORM_DETECTED 1)
    set(OS_IS_WIN 1)
    set(CMAKE_INSTALL_SYSTEM_RUNTIME_LIBS_SKIP TRUE)
endif()
set(CMAKE_INSTALL_PREFIX "${CADENCE_STAGE_ROOT}" CACHE PATH "Existing verified stage" FORCE)
set(MUE_PACK_INSTALL_DIR ON CACHE BOOL "Package the existing stage without installation rules" FORCE)
set(CPACK_WIX_TEMPLATE "${CMAKE_CURRENT_SOURCE_DIR}/buildscripts/packaging/Windows/Installer/WIX.template.in")
include("${CADENCE_SOURCE_ROOT}/buildscripts/packaging/Windows/SetupWindowsPackaging.cmake")
if(NOT CPACK_PACKAGE_VERSION STREQUAL "4.7.6.2002" OR CPACK_INSTALL_CMAKE_PROJECTS)
    message(FATAL_ERROR "Unexpected MSI version or install rules; refusing to alter the retained payload")
endif()
