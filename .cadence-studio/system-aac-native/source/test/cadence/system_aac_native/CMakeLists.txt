cmake_minimum_required(VERSION 3.22)
project(cadence_production_system_aac_test LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This native test requires Windows x64, MSVC and the Windows SDK.")
endif()
get_filename_component(CADENCE_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(AAC_SOURCE "${CADENCE_SOURCE}/src/framework/audio/engine/internal/export")
add_executable(system-aac-native native.cpp "${AAC_SOURCE}/windowsaacstream.cpp")
target_include_directories(system-aac-native PRIVATE "${AAC_SOURCE}")
target_compile_features(system-aac-native PRIVATE cxx_std_17)
target_compile_options(system-aac-native PRIVATE /W4 /EHsc /utf-8)
target_compile_definitions(system-aac-native PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_link_libraries(system-aac-native PRIVATE ole32 mfuuid wmcodecdspuuid)
set_property(TARGET system-aac-native PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
enable_testing()
foreach(rate IN ITEMS 44100 48000)
    add_test(NAME production-system-aac-${rate}
        COMMAND system-aac-native --rate ${rate} --output-root "${CMAKE_CURRENT_BINARY_DIR}/native-results")
    set_tests_properties(production-system-aac-${rate} PROPERTIES TIMEOUT 30)
endforeach()
