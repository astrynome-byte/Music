cmake_minimum_required(VERSION 3.22)
project(CadenceWatchModeStress LANGUAGES CXX)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core)
find_package(Threads REQUIRED)
add_executable(cadence_watch_mode_stress watch_mode_stress.cpp source/src/project/internal/localscorelibrarystore.cpp)
target_include_directories(cadence_watch_mode_stress PRIVATE source/src/project/internal "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(cadence_watch_mode_stress PRIVATE cxx_std_17)
target_link_libraries(cadence_watch_mode_stress PRIVATE Qt6::Core Threads::Threads)
if(WIN32)
    target_link_libraries(cadence_watch_mode_stress PRIVATE advapi32)
endif()
if(MSVC)
    target_compile_options(cadence_watch_mode_stress PRIVATE /utf-8 /W4)
else()
    target_compile_options(cadence_watch_mode_stress PRIVATE -Wall -Wextra -Werror)
endif()

# Exact candidate production store tests; this does not build the application.
add_subdirectory(source/test/cadence/local_library_store production-store-tests)
